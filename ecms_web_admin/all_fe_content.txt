    1: import { useEffect, useState } from 'react';
    2: 
    3: export const useDebounce = <T,>(value: T, delay = 500): T => {
    4:   const [debounced, setDebounced] = useState(value);
    5: 
    6:   useEffect(() => {
    7:     const timer = setTimeout(() => setDebounced(value), delay);
    8:     return () => clearTimeout(timer);
    9:   }, [value, delay]);
   10: 
   11:   return debounced;
   12: };

    1: import axios from 'axios';
    2: import type { AxiosError, AxiosInstance, InternalAxiosRequestConfig } from 'axios';
    3: import { API_ENDPOINT } from '@/constants/api-endpoint';

    5: export const API_HOST = import.meta.env.VITE_API_URL || 'http://localhost:8080';
    6: export const API_BASE_URL = API_HOST;
    7: 
    8: export const APP_CONFIG = {
    9:   API_BASE_URL: API_HOST,
   10:   API_TIMEOUT: 30000,
   11: } as const;
   12: 
   13: interface QueueItem {
   14:   resolve: (value?: unknown) => void;
   15:   reject: (reason?: unknown) => void;
   16: }
   17: 
   18: type RetryConfig = InternalAxiosRequestConfig & { _retry?: boolean };
   19: 
   20: const api: AxiosInstance = axios.create({
   21:   baseURL: API_BASE_URL,
   22:   timeout: APP_CONFIG.API_TIMEOUT,
   23:   withCredentials: true,
   24:   headers: { 'Content-Type': 'application/json' },
   25: });
   26: 
   27: api.interceptors.request.use((config: InternalAxiosRequestConfig) => {
   28:   if (config.data instanceof FormData) {
   29:     delete config.headers['Content-Type'];
   30:   }
   31:   return config;
   32: });
   33: 
   34: let isRefreshing = false;
   35: let failedQueue: QueueItem[] = [];
   36: 
   37: const DEBUG = import.meta.env.DEV;
   38: const log = (...args: unknown[]) => DEBUG && console.log(...args);
   39: 
   40: const NO_REFRESH_URLS: string[] = [
   41:   API_ENDPOINT.AUTH_LOGIN,
   42:   API_ENDPOINT.AUTH_REGISTER,
   43:   API_ENDPOINT.AUTH_REFRESH_TOKEN,
   44:   API_ENDPOINT.AUTH_LOGOUT,
   45:   API_ENDPOINT.AUTH_FORGOT_PASSWORD,
   46:   API_ENDPOINT.AUTH_VERIFY_OTP,
   47:   API_ENDPOINT.AUTH_RESET_PASSWORD,
   48: ];
   49: 
   50: const SILENT_URLS: string[] = [API_ENDPOINT.AUTH_REFRESH_TOKEN];
   51: 
   52: type SessionExpiredListener = () => void;
   53: const sessionExpiredListeners = new Set<SessionExpiredListener>();
   54: 
   55: export const subscribeSessionExpired = (listener: SessionExpiredListener): (() => void) => {
   56:   sessionExpiredListeners.add(listener);
   57:   return () => sessionExpiredListeners.delete(listener);
   58: };
   59: 
   60: const emitSessionExpired = () => {
   61:   sessionExpiredListeners.forEach((fn) => {
   62:     try { fn(); } catch (e) { console.warn('[axiosConfig]', e); }
   63:   });
   64: };
   65: 
   66: const processQueue = (error: unknown) => {
   67:   failedQueue.forEach((p) => (error ? p.reject(error) : p.resolve()));
   68:   failedQueue = [];
   69: };
   70: 
   71: const extractErrorMessage = (error: AxiosError): string => {
   72:   if (!error.response) return 'Không thể kết nối đến server.';
   73:   const data = error.response.data as Record<string, unknown> | undefined;
   74:   if (data?.message) return String(data.message);
   75:   if (data?.error) return String(data.error);
   76: 
   77:   const messages: Record<number, string> = {
   78:     400: 'Yêu cầu không hợp lệ',
   79:     401: 'Phiên đăng nhập đã hết hạn',
   80:     403: 'Bạn không có quyền truy cập',
   81:     404: 'Không tìm thấy tài nguyên',
   82:     409: 'Dữ liệu đã tồn tại',
   83:     423: 'Tài khoản đang bị khóa',
   84:     500: 'Lỗi hệ thống',
   85:   };
   86:   return messages[error.response.status] || 'Có lỗi xảy ra';
   87: };
   88: 
   89: const shouldSkipRefresh = (url?: string): boolean =>
   90:   url ? NO_REFRESH_URLS.some((e) => url.includes(e)) : false;
   91: 
   92: const shouldBeSilent = (url?: string): boolean =>
   93:   url ? SILENT_URLS.some((e) => url.includes(e)) : false;
   94: 
   95: api.interceptors.request.use(
   96:   (config) => {
   97:     if (!shouldBeSilent(config.url)) log('🔵', config.method?.toUpperCase(), config.url);
   98:     return config;
   99:   },
  100:   (error) => Promise.reject(error)
  101: );
  102: 
  103: api.interceptors.response.use(
  104:   (response) => {
  105:     if (!shouldBeSilent(response.config.url)) log('🟢', response.status, response.config.url);
  106: 
  107:     const data = response.data as Record<string, unknown> | undefined;
  108:     if (data?.status === 'error') {
  109:       const err = new Error(String(data.message || 'Có lỗi xảy ra')) as Error & {
  110:         response: unknown;
  111:         isBusinessError: boolean;
  112:         isSilent: boolean;
  113:       };
  114:       err.response = response;
  115:       err.isBusinessError = true;
  116:       err.isSilent = shouldBeSilent(response.config.url);
  117:       return Promise.reject(err);
  118:     }
  119:     return response;
  120:   },
  121:   async (error: AxiosError) => {
  122:     const originalRequest = error.config as RetryConfig | undefined;
  123:     const silent = shouldBeSilent(originalRequest?.url);
  124: 
  125:     if (error.response) error.message = extractErrorMessage(error);
  126: 
  127:     if (silent && error.response?.status === 401) {
  128:       Object.assign(error, { isSilent: true, isAuthError: true, isSessionExpired: true });
  129:       emitSessionExpired();
  130:       return Promise.reject(error);
  131:     }
  132: 
  133:     if (error.response?.status !== 401 || !originalRequest || originalRequest._retry) {
  134:       return Promise.reject(error);
  135:     }
  136: 
  137:     if (shouldSkipRefresh(originalRequest.url)) return Promise.reject(error);
  138: 
  139:     if (isRefreshing) {
  140:       return new Promise<void>((resolve, reject) => {
  141:         failedQueue.push({ resolve, reject });
  142:       }).then(() => api(originalRequest));
  143:     }
  144: 
  145:     originalRequest._retry = true;
  146:     isRefreshing = true;
  147: 
  148:     try {
  149:       await api.post(API_ENDPOINT.AUTH_REFRESH_TOKEN);
  150:       log('✅ Refresh thành công');
  151:       processQueue(null);
  152:       isRefreshing = false;
  153:       return api(originalRequest);
  154:     } catch (refreshError: unknown) {
  155:       const err = refreshError as AxiosError;
  156:       if (err?.response?.status === 401) {
  157:         Object.assign(err, { isSilent: true, isAuthError: true, isSessionExpired: true });
  158:         emitSessionExpired();
  159:       }
  160:       processQueue(refreshError);
  161:       isRefreshing = false;
  162:       return Promise.reject(refreshError);
  163:     }
  164:   }
  165: );
  166: 
  167: export default api;
